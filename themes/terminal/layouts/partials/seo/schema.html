:= partial "seo/canonical.html" . -}} {{- $title := partial "seo/title.html" . -}} {{- $description := partial "seo/description.html" . -}} {{- $card := partialCached "og/image.html" . .RelPermalink -}} {{- $sameAs := slice -}} {{- with site.Params.TwitterHandle -}} {{- $sameAs = $sameAs | append (printf "https://twitter.com/%s" .) -}} {{- end -}} {{- with site.Params.GitHubUser -}} {{- $sameAs = $sameAs | append (printf "https://github.com/%s" .) -}} {{- end -}} {{- $personId := printf "%s#person" $home -}} {{- $siteId := printf "%s#website" $home -}} {{- $person := dict "@type" "Person" "inLanguage" "en-GB" "publisher" (dict "@id" $personId) -}} {{- $graph := slice $person $website -}} {{- if and .IsPage (eq .Section "blog") -}} {{- $article := dict "@type" "BlogPosting" "@id" .Permalink "width" .Width "height" .Height)) -}} {{- end -}} {{- if not .Date.IsZero -}} {{- $article = merge $article (dict "datePublished" (.Date.Format "2006-01-02T15:04:05Z07:00")) -}} {{- end -}} (dict "keywords" .) -}} {{- end -}} {{- $graph = $graph | append $article -}} {{- $crumbs := slice (dict "@type" "ListItem" "position" 1 "name" "Home" "item" $home) (dict "@type" "ListItem" "position"
